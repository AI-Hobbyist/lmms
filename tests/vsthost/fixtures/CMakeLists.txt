cmake_minimum_required(VERSION 3.13)
project(LmmsVstFixtures LANGUAGES CXX)
add_library(Vst2Baseline MODULE Vst2Baseline.cpp)
target_compile_features(Vst2Baseline PRIVATE cxx_std_20)
target_include_directories(Vst2Baseline PRIVATE ../../../include)
set_target_properties(Vst2Baseline PROPERTIES PREFIX "")

if(WIN32)
	foreach(shell_mode Normal Hang Crash Duplicate)
		if(shell_mode STREQUAL "Normal")
			set(shell_target Vst2Shell)
		else()
			set(shell_target Vst2Shell${shell_mode})
		endif()
		add_library(${shell_target} MODULE Vst2Shell.cpp)
		target_compile_features(${shell_target} PRIVATE cxx_std_20)
		target_include_directories(${shell_target} PRIVATE ../../../include)
		set_target_properties(${shell_target} PROPERTIES PREFIX "")
		if(NOT shell_mode STREQUAL "Normal")
			string(TOUPPER "${shell_mode}" shell_definition)
			target_compile_definitions(${shell_target} PRIVATE VST_SHELL_SCAN_${shell_definition})
		endif()
	endforeach()
	set(vst_fault_operations Entry Audio State Editor Close)
	set(vst_fault_operation 0)
	foreach(operation IN LISTS vst_fault_operations)
		foreach(mode Crash Hang)
			set(target Vst2Fault${operation}${mode})
			add_library(${target} MODULE Vst2Fault.cpp)
			target_compile_features(${target} PRIVATE cxx_std_20)
			target_include_directories(${target} PRIVATE ../../../include)
			set_target_properties(${target} PROPERTIES PREFIX "")
			if(mode STREQUAL "Hang")
				set(vst_fault_hang 1)
			else()
				set(vst_fault_hang 0)
			endif()
			target_compile_definitions(${target} PRIVATE VST_FAULT_OPERATION=${vst_fault_operation} VST_FAULT_HANG=${vst_fault_hang})
		endforeach()
		math(EXPR vst_fault_operation "${vst_fault_operation} + 1")
	endforeach()
endif()
