cmake_minimum_required(VERSION 3.16)
project(SVSConformance LANGUAGES CXX)
if(NOT TARGET SVSSDK::SDK)
	find_package(SVSSDK CONFIG REQUIRED)
endif()
add_executable(SVSConformance SVSConformance.cpp)
target_compile_features(SVSConformance PRIVATE cxx_std_17)
target_link_libraries(SVSConformance PRIVATE SVSSDK::SDK ${CMAKE_DL_LIBS})
if(DEFINED SVS_JSON_INCLUDE)
	target_include_directories(SVSConformance PRIVATE "${SVS_JSON_INCLUDE}")
endif()
install(TARGETS SVSConformance RUNTIME DESTINATION bin)
