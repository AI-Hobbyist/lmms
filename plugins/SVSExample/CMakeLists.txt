cmake_minimum_required(VERSION 3.13)
project(SVSExample LANGUAGES CXX)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/full.json" SVS_FULL_SCHEMA)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/minimal.json" SVS_MINIMAL_SCHEMA)
configure_file(Schemas.h.in "${CMAKE_CURRENT_BINARY_DIR}/Schemas.h" @ONLY)
set(SVS_SDK_INCLUDE "${CMAKE_CURRENT_SOURCE_DIR}/../../sdk/svs/include" CACHE PATH "Installed SVS SDK include directory")
add_library(SVSExample MODULE SVSExample.cpp)
target_include_directories(SVSExample PRIVATE "${SVS_SDK_INCLUDE}" "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_features(SVSExample PRIVATE cxx_std_17)
target_compile_definitions(SVSExample PRIVATE SVS_PLUGIN_BUILD)
set_target_properties(SVSExample PROPERTIES PREFIX "" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/svs/SVSExample/$<0:unused>" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/svs/SVSExample/$<0:unused>")
foreach(resource manifest.json avatar.svg portrait.svg zh.json ja.json en.json)
  add_custom_command(TARGET SVSExample POST_BUILD COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/${resource}" "$<TARGET_FILE_DIR:SVSExample>/${resource}")
endforeach()
install(TARGETS SVSExample LIBRARY DESTINATION svs/SVSExample RUNTIME DESTINATION svs/SVSExample)
install(FILES manifest.json avatar.svg portrait.svg zh.json ja.json en.json DESTINATION svs/SVSExample)
